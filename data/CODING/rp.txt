<script>
(function () {
  const root = document.getElementById("pythonLessonRoot");
  if (!root) return;

  const shell = root.querySelector(".python-lesson-shell");
  const runnerColumn = root.querySelector(".python-runner-column");

  if (!shell || !runnerColumn) return;

  function positionRunner() {
    if (!shell.isConnected || !runnerColumn.isConnected) return;

    const rect = shell.getBoundingClientRect();
    const viewportWidth = document.documentElement.clientWidth;
    const inset = viewportWidth <= 600 ? 6 : 16;

    let left = Math.max(6, rect.left + inset);
    let right = Math.min(viewportWidth - 6, rect.right - inset);
    let width = right - left;

    if (width < 260) {
      left = 6;
      width = viewportWidth - 12;
    }

    runnerColumn.style.left = Math.round(left) + "px";
    runnerColumn.style.width = Math.round(width) + "px";
    runnerColumn.style.transform = "none";
  }

  // 렌더링 후 최초 위치 잡기
  requestAnimationFrame(positionRunner);

  // 브라우저 크기 변경 시 대응
  const resizeHandler = () => positionRunner();
  window.addEventListener("resize", resizeHandler, { passive: true });

  let resizeObserver = null;
  if (typeof ResizeObserver !== "undefined") {
    resizeObserver = new ResizeObserver(() => positionRunner());
    resizeObserver.observe(shell);
  }
})();
