14강. Vue 3 전역 상태 관리 (Pinia & Setup Store)

1. 왜 Pinia인가? (가장 모던한 Vue 3 공식 전역 상태 관리)

🍍 Pinia는 Vuex를 대체하는 Vue 3 공식 전역 상태 관리 라이브러리입니다.
  • Setup Store 문법: ref()가 State, computed()가 Getters, 일반 함수가 Actions가 되어 <script setup>과 100% 동일한 문법으로 작성합니다.
  • TypeScript 완벽 지원 및 가벼운 번들 크기(~1KB), DevTools 타임트래블 디버깅을 지원합니다.

2. 실무 Pinia 코드 구조 (stores/useUserStore.js)

실무 Pinia Setup Store 예시 (참고용) 
// stores/useUserStore.js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useUserStore = defineStore('user', () => {
  const userName = ref('게스트')
  const cartCount = ref(0)
  const isLoggedIn = computed(() => userName.value !== '게스트')

  function login(name) { userName.value = name }
  function addCart() { cartCount.value++ }

  return { userName, cartCount, isLoggedIn, login, addCart }
})

3. 기본 뼈대 코드 (Starter Code - 샌드박스 즉시 실행)

Starter Code (기본 뼈대) 
<script setup>
import { ref, computed } from 'vue'

// Pinia Setup Store 패턴을 단일 파일에서 체험
function createGlobalStore() {
  const userName = ref('게스트')
  const cartCount = ref(0)
  const isLoggedIn = computed(() => userName.value !== '게스트')
  const login = (name) => { userName.value = name }
  const addCart = () => { cartCount.value++ }
  return { userName, cartCount, isLoggedIn, login, addCart }
}

const store = createGlobalStore()
</script>

<template>
  <div>
    <h2>🍍 {{ store.userName }} (장바구니: {{ store.cartCount }})</h2>
    <button @click="store.login('김개발')">로그인</button>
    <button @click="store.addCart()">상품 담기</button>
  </div>
</template>

4. 완성 정답 코드 (Solution Code)

Solution Code (완성 정답) 
<script setup>
import { ref, computed } from 'vue'

// Pinia defineStore('app', () => { ... }) 와 동일한 Setup Store 패턴
const useAppStore = (() => {
  let instance = null
  return () => {
    if (instance) return instance
    const user = ref({ name: '게스트', role: 'visitor' })
    const notifications = ref(['Pinia 스토어에 오신 것을 환영합니다!'])
    const theme = ref('light')

    const isLoggedIn = computed(() => user.value.name !== '게스트')
    const unreadCount = computed(() => notifications.value.length)

    const login = (name) => {
      if (!name.trim()) return
      user.value = { name, role: 'admin' }
      notifications.value.unshift(`${name}님이 로그인했습니다.`)
    }
    const logout = () => {
      user.value = { name: '게스트', role: 'visitor' }
      notifications.value.unshift('로그아웃 되었습니다.')
    }
    const toggleTheme = () => {
      theme.value = theme.value === 'light' ? 'dark' : 'light'
    }

    instance = { user, notifications, theme, isLoggedIn, unreadCount, login, logout, toggleTheme }
    return instance
  }
})()

// 여러 영역(헤더, 본문, 알림창)이 동일한 전역 스토어를 공유
const headerStore = useAppStore()
const bodyStore = useAppStore()
const inputName = ref('')
</script>

<template>
  <div :class="['app-wrap', headerStore.theme]">
    <header class="top-bar">
      <strong>🍍 Pinia 전역 상태 데모</strong>
      <div class="top-right">
        <span>🔔 {{ headerStore.unreadCount }}</span>
        <span>👤 {{ headerStore.user.value.name }}</span>
        <button @click="headerStore.toggleTheme()">
          {{ headerStore.theme === 'light' ? '🌙 다크' : '☀️ 라이트' }}
        </button>
      </div>
    </header>

    <section class="panel">
      <div v-if="!bodyStore.isLoggedIn.value" class="login-row">
        <input v-model="inputName" placeholder="이름 입력 후 로그인" @keyup.enter="bodyStore.login(inputName)" />
        <button class="btn-primary" @click="bodyStore.login(inputName)">전역 로그인</button>
      </div>
      <div v-else class="login-row">
        <span>현재 <strong>{{ bodyStore.user.value.name }}</strong> ({{ bodyStore.user.value.role }}) 로 로그인됨</span>
        <button class="btn-out" @click="bodyStore.logout()">로그아웃</button>
      </div>
    </section>

    <section class="panel">
      <h3>전역 알림 로그</h3>
      <ul class="noti-list">
        <li v-for="(msg, i) in bodyStore.notifications.value" :key="i">{{ msg }}</li>
      </ul>
    </section>
  </div>
</template>
