3강. 우선순위 할 일 & 필터링 대시보드

1. 완성 화면 요구사항 & 핵심 포인트

  • 완료율 계산(computed): 전체 할 일 중 완료된 비율(%)을 계산해 진행률 바를 표시합니다.
  • 상태별 탭 전환: 전체 / 진행중 / 완료 탭으로 필터링합니다.
  • 로컬 스토리지 연동(watch & onMounted): 할 일 목록이 바뀔 때마다 localStorage에 자동 저장하고, 새로고침 시 복원합니다.

2. 기본 뼈대 코드 (Starter Code)

Starter Code (기본 뼈대) 
<script setup>
import { ref, computed, watch, onMounted } from 'vue'

const todos = ref([])
const newText = ref('')
const priority = ref('보통')

const addTodo = () => {
  if (!newText.value.trim()) return
  todos.value.push({ id: Date.now(), text: newText.value, priority: priority.value, done: false })
  newText.value = ''
}
</script>

<template>
  <div>
    <h2>✅ 우선순위 할 일</h2>
    <input v-model="newText" placeholder="할 일 입력" @keyup.enter="addTodo" />
    <button @click="addTodo">추가</button>
    <ul>
      <li v-for="t in todos" :key="t.id">{{ t.text }} ({{ t.priority }})</li>
    </ul>
  </div>
</template>

3. 완성 정답 코드 (Solution Code)

Solution Code (완성 정답) 
<script setup>
import { ref, computed, watch, onMounted } from 'vue'

const STORAGE_KEY = 'vue3_todo_dashboard'
const todos = ref([])
const newText = ref('')
const priority = ref('보통')
const filterTab = ref('all')

onMounted(() => {
  const saved = localStorage.getItem(STORAGE_KEY)
  if (saved) {
    todos.value = JSON.parse(saved)
  } else {
    todos.value = [
      { id: 1, text: 'Vue 3 computed 복습하기', priority: '높음', done: true },
      { id: 2, text: '미니 프로젝트 완성하기', priority: '보통', done: false }
    ]
  }
})

watch(todos, (val) => {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(val))
}, { deep: true })

const completionRate = computed(() => {
  if (todos.value.length === 0) return 0
  const doneCount = todos.value.filter(t => t.done).length
  return Math.round((doneCount / todos.value.length) * 100)
})

const filteredTodos = computed(() => {
  if (filterTab.value === 'active') return todos.value.filter(t => !t.done)
  if (filterTab.value === 'done') return todos.value.filter(t => t.done)
  return todos.value
})

const addTodo = () => {
  if (!newText.value.trim()) return
  todos.value.unshift({ id: Date.now(), text: newText.value, priority: priority.value, done: false })
  newText.value = ''
}

const removeTodo = (id) => {
  todos.value = todos.value.filter(t => t.id !== id)
}
</script>

<template>
  <div class="todo-app">
    <h2>✅ 우선순위 할 일 대시보드</h2>
    <div class="progress-box">
      <div class="progress-label">완료율: {{ completionRate }}%</div>
      <div class="progress-bar">
        <div class="progress-fill" :style="{ width: completionRate + '%' }"></div>
      </div>
    </div>

    <div class="input-row">
      <select v-model="priority">
        <option>높음</option><option>보통</option><option>낮음</option>
      </select>
      <input v-model="newText" placeholder="할 일을 입력하세요..." @keyup.enter="addTodo" />
      <button @click="addTodo">추가</button>
    </div>

    <div class="tabs">
      <button :class="{ active: filterTab === 'all' }" @click="filterTab = 'all'">전체</button>
      <button :class="{ active: filterTab === 'active' }" @click="filterTab = 'active'">진행중</button>
      <button :class="{ active: filterTab === 'done' }" @click="filterTab = 'done'">완료</button>
    </div>

    <ul class="list">
      <li v-for="t in filteredTodos" :key="t.id" :class="{ done: t.done }">
        <label>
          <input type="checkbox" v-model="t.done" />
          <span class="tag" :class="t.priority">{{ t.priority }}</span>
          <span>{{ t.text }}</span>
        </label>
        <button class="del" @click="removeTodo(t.id)">✕</button>
      </li>
    </ul>
  </div>
</template>
