7강. 스터디 타이머 & 랩타임 기록기

1. 완성 화면 요구사항 & 핵심 포인트

  • 타이머 제어(setInterval): 시작 / 일시정지 / 초기화 기능을 구현합니다.
  • 랩타임 누적(v-for): 현재 경과 시간을 포맷팅(MM:SS)하여 랩타임 배열에 기록합니다.
  • 메모리 누수 방지(onUnmounted): 컴포넌트가 사라질 때 clearInterval로 타이머를 반드시 정리합니다.

2. 기본 뼈대 코드 (Starter Code)

Starter Code (기본 뼈대) 
<script setup>
import { ref, onUnmounted } from 'vue'

const seconds = ref(0)
const isRunning = ref(false)
let timerId = null

const start = () => {
  if (isRunning.value) return
  isRunning.value = true
  timerId = setInterval(() => seconds.value++, 1000)
}
onUnmounted(() => clearInterval(timerId))
</script>

<template>
  <div>
    <h2>⏱️ {{ seconds }}초</h2>
    <button @click="start">시작</button>
  </div>
</template>

3. 완성 정답 코드 (Solution Code)

Solution Code (완성 정답) 
<script setup>
import { ref, computed, onUnmounted } from 'vue'

const elapsed = ref(0)
const isRunning = ref(false)
const laps = ref([])
let timerId = null

const formattedTime = computed(() => {
  const mins = String(Math.floor(elapsed.value / 60)).padStart(2, '0')
  const secs = String(elapsed.value % 60).padStart(2, '0')
  return `${mins}:${secs}`
})

const start = () => {
  if (isRunning.value) return
  isRunning.value = true
  timerId = setInterval(() => {
    elapsed.value++
  }, 1000)
}

const pause = () => {
  isRunning.value = false
  clearInterval(timerId)
}

const reset = () => {
  pause()
  elapsed.value = 0
  laps.value = []
}

const recordLap = () => {
  laps.value.unshift({ id: Date.now(), time: formattedTime.value })
}

onUnmounted(() => {
  clearInterval(timerId)
})
</script>

<template>
  <div class="timer-box">
    <h2>⏱️ 스터디 타이머</h2>
    <div class="clock">{{ formattedTime }}</div>
    <div class="btns">
      <button v-if="!isRunning" class="start" @click="start">▶ 시작</button>
      <button v-else class="pause" @click="pause">⏸ 일시정지</button>
      <button class="lap" @click="recordLap" :disabled="elapsed === 0">🏁 랩 기록</button>
      <button class="reset" @click="reset">↺ 초기화</button>
    </div>

    <ul class="laps" v-if="laps.length > 0">
      <li v-for="(lap, idx) in laps" :key="lap.id">
        <span>Lap {{ laps.length - idx }}</span>
        <strong>{{ lap.time }}</strong>
      </li>
    </ul>
  </div>
</template>
