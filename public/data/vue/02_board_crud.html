2강. 게시판 화면 (검색 + 모달 CRUD)

1. 완성 화면 요구사항 & 핵심 포인트

  • 목록 렌더링(v-for): 게시글 배열을 테이블로 출력합니다.
  • 검색 & 카테고리 필터(computed): 검색어와 카테고리 선택값에 따라 실시간으로 게시글을 필터링합니다.
  • 모달 제어(v-show): 글쓰기 모달과 상세보기 모달을 v-show로 열고 닫으며, 게시글 추가 및 삭제를 구현합니다.

2. 기본 뼈대 코드 (Starter Code)

Starter Code (기본 뼈대) 
<script setup>
import { ref, computed } from 'vue'

const posts = ref([
  { id: 1, category: '공지', title: 'Vue 3 스터디 안내', content: '매주 실습을 진행합니다.' },
  { id: 2, category: '질문', title: 'v-if와 v-show 차이?', content: '언제 무엇을 쓰나요?' }
])
const search = ref('')
const selectedCategory = ref('전체')

const filteredPosts = computed(() => posts.value)
</script>

<template>
  <div class="board">
    <h2>📋 미니 게시판</h2>
    <input v-model="search" placeholder="제목 검색..." />
    <ul>
      <li v-for="p in filteredPosts" :key="p.id">[{{ p.category }}] {{ p.title }}</li>
    </ul>
  </div>
</template>

3. 완성 정답 코드 (Solution Code)

Solution Code (완성 정답) 
<script setup>
import { ref, computed } from 'vue'

const posts = ref([
  { id: 1, category: '공지', title: 'Vue 3 실전 커리큘럼 오픈', content: '1강부터 10강까지 직접 만들어보세요!' },
  { id: 2, category: '질문', title: 'computed는 언제 쓰나요?', content: '파생 데이터를 캐싱할 때 사용합니다.' },
  { id: 3, category: '자유', title: '오늘의 커피 추천', content: '아이스 아메리카노 최고!' }
])

const search = ref('')
const selectedCategory = ref('전체')
const showWriteModal = ref(false)
const detailPost = ref(null)

const newTitle = ref('')
const newCategory = ref('자유')
const newContent = ref('')

const filteredPosts = computed(() => {
  return posts.value.filter(p => {
    const matchCat = selectedCategory.value === '전체' || p.category === selectedCategory.value
    const matchText = p.title.toLowerCase().includes(search.value.toLowerCase())
    return matchCat && matchText
  })
})

const addPost = () => {
  if (!newTitle.value.trim()) return
  posts.value.unshift({
    id: Date.now(),
    category: newCategory.value,
    title: newTitle.value,
    content: newContent.value || '내용 없음'
  })
  newTitle.value = ''
  newContent.value = ''
  showWriteModal.value = false
}

const deletePost = (id) => {
  posts.value = posts.value.filter(p => p.id !== id)
  if (detailPost.value?.id === id) detailPost.value = null
}
</script>

<template>
  <div class="board">
    <div class="top-bar">
      <h2>📋 미니 게시판</h2>
      <button class="btn-primary" @click="showWriteModal = true">+ 글쓰기</button>
    </div>

    <div class="filters">
      <select v-model="selectedCategory">
        <option>전체</option><option>공지</option><option>질문</option><option>자유</option>
      </select>
      <input v-model="search" placeholder="제목 검색..." />
    </div>

    <table>
      <thead><tr><th>분류</th><th>제목</th><th>관리</th></tr></thead>
      <tbody>
        <tr v-for="p in filteredPosts" :key="p.id">
          <td><span class="badge">{{ p.category }}</span></td>
          <td class="title-cell" @click="detailPost = p">{{ p.title }}</td>
          <td><button class="btn-del" @click="deletePost(p.id)">삭제</button></td>
        </tr>
      </tbody>
    </table>

    <div v-show="showWriteModal" class="modal-backdrop">
      <div class="modal">
        <h3>새 글 작성</h3>
        <select v-model="newCategory"><option>공지</option><option>질문</option><option>자유</option></select>
        <input v-model="newTitle" placeholder="제목" />
        <textarea v-model="newContent" placeholder="내용"></textarea>
        <div class="modal-actions">
          <button class="btn-primary" @click="addPost">등록</button>
          <button @click="showWriteModal = false">취소</button>
        </div>
      </div>
    </div>

    <div v-show="detailPost" class="modal-backdrop">
      <div class="modal" v-if="detailPost">
        <h3>[{{ detailPost.category }}] {{ detailPost.title }}</h3>
        <p>{{ detailPost.content }}</p>
        <button @click="detailPost = null">닫기</button>
      </div>
    </div>
  </div>
</template>
