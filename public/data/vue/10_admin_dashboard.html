10강. 종합 미니 관리자 페이지 (Admin Dashboard)

1. 완성 화면 요구사항 & 핵심 포인트

  • 상단 요약 카드(computed): 전체 주문 수, 처리 대기, 배송 완료 건수를 실시간 집계합니다.
  • 검색 조건 폼: 주문자명 검색 + 상태 필터 드롭다운을 조합합니다.
  • 상태 변경 & 상세 수정 팝업: 테이블 내 드롭다운으로 상태를 즉시 변경하고, [상세] 버튼 클릭 시 모달에서 메모를 수정합니다.

2. 기본 뼈대 코드 (Starter Code)

Starter Code (기본 뼈대) 
<script setup>
import { ref, computed } from 'vue'

const orders = ref([
  { id: 'ORD-01', customer: '김철수', status: '대기', amount: 45000, memo: '' },
  { id: 'ORD-02', customer: '이영희', status: '완료', amount: 89000, memo: '문 앞 배송' }
])
const search = ref('')
const filtered = computed(() => orders.value.filter(o => o.customer.includes(search.value)))
</script>

<template>
  <div>
    <h2>🛠️ 미니 관리자 대시보드</h2>
    <input v-model="search" placeholder="주문자 검색" />
    <ul><li v-for="o in filtered" :key="o.id">{{ o.customer }} ({{ o.status }})</li></ul>
  </div>
</template>

3. 완성 정답 코드 (Solution Code)

Solution Code (완성 정답) 
<script setup>
import { ref, computed } from 'vue'

const orders = ref([
  { id: 'ORD-101', customer: '김철수', status: '대기', amount: 45000, memo: '빠른 배송 요청' },
  { id: 'ORD-102', customer: '이영희', status: '배송중', amount: 89000, memo: '경비실 보관' },
  { id: 'ORD-103', customer: '박민수', status: '완료', amount: 32000, memo: '' }
])

const search = ref('')
const statusFilter = ref('전체')
const editingOrder = ref(null)

const pendingCount = computed(() => orders.value.filter(o => o.status === '대기').length)
const doneCount = computed(() => orders.value.filter(o => o.status === '완료').length)

const filteredOrders = computed(() => {
  return orders.value.filter(o => {
    const matchName = o.customer.includes(search.value) || o.id.includes(search.value)
    const matchStatus = statusFilter.value === '전체' || o.status === statusFilter.value
    return matchName && matchStatus
  })
})
</script>

<template>
  <div class="admin">
    <h2>🛠️ 종합 미니 관리자 대시보드</h2>
    <div class="kpi-grid">
      <div class="kpi">전체 주문<strong>{{ orders.length }}건</strong></div>
      <div class="kpi warn">처리 대기<strong>{{ pendingCount }}건</strong></div>
      <div class="kpi ok">배송 완료<strong>{{ doneCount }}건</strong></div>
    </div>

    <div class="search-bar">
      <input v-model="search" placeholder="주문번호 또는 고객명..." />
      <select v-model="statusFilter">
        <option>전체</option><option>대기</option><option>배송중</option><option>완료</option>
      </select>
    </div>

    <table>
      <thead><tr><th>주문번호</th><th>고객명</th><th>금액</th><th>상태</th><th>상세</th></tr></thead>
      <tbody>
        <tr v-for="o in filteredOrders" :key="o.id">
          <td>{{ o.id }}</td>
          <td>{{ o.customer }}</td>
          <td>{{ o.amount.toLocaleString() }}원</td>
          <td>
            <select v-model="o.status">
              <option>대기</option><option>배송중</option><option>완료</option>
            </select>
          </td>
          <td><button class="btn-edit" @click="editingOrder = o">수정</button></td>
        </tr>
      </tbody>
    </table>

    <div v-show="editingOrder" class="modal-bg">
      <div class="modal" v-if="editingOrder">
        <h3>주문 상세 ({{ editingOrder.id }})</h3>
        <p>고객명: {{ editingOrder.customer }}</p>
        <label>관리자 메모:</label>
        <textarea v-model="editingOrder.memo" rows="3"></textarea>
        <button class="btn-close" @click="editingOrder = null">저장 및 닫기</button>
      </div>
    </div>
  </div>
</template>
